gom <- read.csv("../tax_annotation_proj/tax_to_annotate.csv") %>% 
  # filter(grepl("Dinoflagellata|Ciliophora", Taxon)) %>% 
  separate(Taxon, into = c("Domain", "Supergroup", "Division", "Subdivision", "Class", "Order", "Family", "Genus", "Species"), sep = ";", remove = FALSE) %>%

joined <- fuzzyjoin::stringdist_left_join(
  gom, jones,
  by = c("GENUS_SPECIES"),
  distance_col = "DIST",
  ignore_case = TRUE,
  method = "lv"
) %>% 
  mutate(Morphotype = (str_replace_all(Morphotype, "eGNCM", "eSNCM"))) #revision based on Mitra et al. mixotrophy database

Summary of feeding modes and morphotypes

table(result_joined %>% select(Subdivision, Feeding.Mode))

table(result_joined %>% select(Subdivision, Morphotype))

CM: Constitutive mixotroph NCM: non-constitutive mixotroph GNCM: generalist non-constitutive mixotroph SNCM: specialist non-constitutive mixotroph

Functional type definitions (Mitra et al.: * CM: Constitutive mixotroph that has a natural ability to photosynthesize. * _CM*_: Evidence for constitutive mixotrophy, but not recorded * eGNCM: Acquires photosynthetic machinery from a variety of prey and retains as endosymbionts. * PA: Photoautotroph * pSNCM: Mixotroph that steals photosynthetic machinery from a specific prey type. plastidic specialist non-constitutive mixotroph. * SH: Strict heterotrophy * _SH*_: Evidence for strict heterotroph, but not recorded
